struct Punto
    var int x, y;
endstruct

struct Triangulo
    var Punto p1;
    var Punto p2;
    var Punto p3;
endstruct

fun int moverPunto(Punto* p, int dx, int dy)
    p.x = p.x + dx;
    p.y = p.y + dy;
    return 0
endfun

fun int trasladarTriangulo(Triangulo* t, int dx, int dy)
    var int r;
    r = moverPunto(&t.p1, dx, dy);
    r = moverPunto(&t.p2, dx, dy);
    r = moverPunto(&t.p3, dx, dy);
    return 0
endfun

fun int sumaCoordenadas(Triangulo* t)
    var int total;
    total = t.p1.x + t.p1.y + t.p2.x + t.p2.y + t.p3.x + t.p3.y;
    return total
endfun

fun int main()
    var Triangulo t;
    var int r;
    t.p1.x = 0;
    t.p1.y = 0;
    t.p2.x = 3;
    t.p2.y = 4;
    t.p3.x = 6;
    t.p3.y = 0;
    print(sumaCoordenadas(&t));
    r = trasladarTriangulo(&t, 5, 10);
    print(t.p1.x);
    print(t.p1.y);
    print(t.p2.x);
    print(t.p2.y);
    print(t.p3.x);
    print(t.p3.y);
    print(sumaCoordenadas(&t))
endfun
