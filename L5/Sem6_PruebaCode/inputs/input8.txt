operacion = 2;
monto = 150;
exito = false;
switch operacion
case 1
    saldo = saldo + monto;
    exito = true;
    break
case 2
    if monto <= saldo then
        saldo = saldo - monto;
        exito = true
    else
        exito = false
    endif;
    break
case 3
    exito = true;
    break
default
    exito = false
endswitch;
